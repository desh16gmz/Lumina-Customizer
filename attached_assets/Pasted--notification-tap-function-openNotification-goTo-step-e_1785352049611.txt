/* ---------- vision board: fills the 3 "focus" slots with the user's uploads.
   The 2 secondary photos (photo.jpeg, photo1.jpeg) are static in the HTML. ---------- */
function populateBoard(photos){
  for(let i=0;i<PHOTO_COUNT;i++){
    const img = document.getElementById(`board-photo-${i}`);
    if(img) img.src = photos[i] || 'photo1.jpeg';
  }
}

/* ---------- letter typewriter ---------- */
let letterTypeTimer = null;
function typewriterLetter(text){
  const textEl = document.getElementById('letter-body-text');
  const cursorEl = document.getElementById('letter-cursor');
  const bodyEl = document.getElementById('letter-body');
  if(!textEl) return;
  if(letterTypeTimer){ clearInterval(letterTypeTimer); letterTypeTimer = null; }
  textEl.textContent = '';
  cursorEl.style.display = 'inline-block';
  const chars = Array.from(text || '');
  let i = 0;
  letterTypeTimer = setInterval(()=>{
    textEl.textContent += chars[i] || '';
    i++;
    bodyEl.scrollTop = bodyEl.scrollHeight;
    if(i >= chars.length){
      clearInterval(letterTypeTimer);
      letterTypeTimer = null;
      cursorEl.style.display = 'none';
    }
  }, 24);
}

/* ---------- clock (decorative, real local time) ---------- */
(function tickClock(){
  const el = document.getElementById('notif-clock');
  function upd(){
    const d = new Date();
    let h = d.getHours()%12; if(h===0) h=12;
    const m = String(d.getMinutes()).padStart(2,'0');
    if(el) el.textContent = `${h}:${m}`;
  }
  upd(); setInterval(upd, 15000);
})();

/* ---------- nav ---------- */
function goTo(id){
  ['step-intake','step-payment','step-linkout','step-notifications','step-envelope','step-gif','step-letter','step-gallery','step-bootload'].forEach(s=>{
    document.getElementById(s).style.display = 'none';
  });
  const el = document.getElementById(id);
  el.style.display = 'flex';
  el.classList.remove('fade-up'); void el.offsetWidth; el.classList.add('fade-up');
  if(id === 'step-letter') typewriterLetter(state.message);
}
function restartFlow(){
  document.getElementById('env-flap').style.transform = 'scaleY(1)';
  document.getElementById('env-hint').textContent = 'tap to open';
  goTo('step-intake');
}
function goBackTo(id){
  // stepping back through the envelope must reset it, otherwise it looks
  // already-opened next time it's shown
  document.getElementById('env-flap').style.transform = 'scaleY(1)';
  document.getElementById('env-hint').textContent = 'tap to open';
  goTo(id);
}

/* ---------- viewer mode: ?id=<row uuid> (new, short) or ?d=<base64 payload> (legacy) ---------- */
function renderViewer(recipient, sender, message, photos){
  isViewerMode = true;
  state.recipient = recipient;
  state.sender = sender;
  state.message = message;
  state.photos = photos;

  document.getElementById('notif-sender-name').textContent = sender;
  document.getElementById('env-title').textContent = `Happy Girlfriend's Day`;
  document.getElementById('letter-name').textContent = recipient;
  document.getElementById('letter-sender').textContent = sender;
  populateBoard(photos);

  document.getElementById('gallery-actions').innerHTML = '<button class="btn-ghost" onclick="goTo(\'step-letter\')">← Back</button>';
  document.getElementById('gallery-restart').style.display = 'none';
  document.getElementById('notif-edit-btn').style.display = 'none';
  document.getElementById('env-edit-btn').style.display = 'none';
  goTo('step-notifications');
}

(async function initViewer(){
  const params = new URLSearchParams(location.search);
  const id = params.get('id');
  const d = params.get('d');

  function bootError(msg){
    document.getElementById('bootload-spinner').style.display = 'none';
    document.getElementById('bootload-text').textContent = msg;
    document.getElementById('bootload-retry').style.display = 'inline-block';
  }

  if(id){
    try{
      const { data, error } = await withRetry(
        ()=> sb.from('girlfriends_day').select('*').eq('id', id).eq('paid', true).single(),
        { attempts: 3, baseDelay: 800 }
      );
      if(error || !data) throw error || new Error('not found');
      document.documentElement.classList.remove('viewer-boot');
      renderViewer(data.recipient, data.sender, data.message, [data.photo1, data.photo2, data.photo3]);
    }catch(err){
      console.error('Could not load surprise', err);
      bootError("Hmm, this link isn't loading. Check your connection and try again.");
    }
    return;
  }

  if(d){
    try{
      const payload = JSON.parse(decodeURIComponent(escape(atob(d))));
      document.documentElement.classList.remove('viewer-boot');
      renderViewer(payload.r, payload.s, payload.m, payload.p);
    }catch(err){
      console.error('Bad link payload', err);
      bootError("This link looks broken. Ask for a new one.");
    }
    return;
  }

  // creator flow — surface any unfinished paid order so it's never silently lost
  if(loadOrderLocal()) document.getElementById('resume-banner').style.display = 'block';
})();
</script>
